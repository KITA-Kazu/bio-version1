(selectedIdx === correctIdx); if (isCorrect) { btn.classList.add('correct'); score += currentLevel * 10; consecutiveCorrect++; consecutiveIncorrect = 0; // 復習モードで正解したらリストから除外 if (isReviewMode) { 次のnextQuestionで0になるよう調整 } } } else { btn.classList.add('incorrect'); // 正解ボタンを強調表示 buttons.forEach(b => { if (b.innerText === qData.options[correctIdx]) { b.classList.add('correct'); } }); consecutiveIncorrect++; consecutiveCorrect = 0; // 間違いノートに保存 if (!wrongQuestionIds.includes(qData.id)) { wrongQuestionIds.push(qData.id); localStorage.setItem('immune_quiz_wrong', JSON.stringify(wrongQuestionIds)); } // レベルダウン判定（3問連続不正解） if (!isReviewMode && consecutiveIncorrect >= 3) { if (currentLevel > 1) { currentLevel--; consecutiveIncorrect = 0; alert(`3問連続不正解...
